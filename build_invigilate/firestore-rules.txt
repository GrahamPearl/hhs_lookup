/**
 * FIRESTORE SECURITY RULES
 * 
 * Copy these rules to your Firebase Console:
 * 1. Go to Firebase Console > yourproject > Firestore Database > Rules
 * 2. Replace the existing rules with the code below
 * 3. Click "Publish"
 * 
 * These rules allow:
 * - Anyone to read and write assignments (for offline-first app)
 * - Automatic timestamp management
 * - Batch operations
 */

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Collection: assignments
    match /assignments/{document=**} {
      // Allow anyone to read and write (for offline-first app)
      // In production, add authentication rules: allow read, write: if request.auth != null;
      allow read, write: if true;
      
      // Validate assignment structure
      function isValidAssignment(data) {
        return data.keys().hasAll(['date', 'session', 'exam', 'educator']) &&
               data.date is string &&
               data.session is number &&
               data.exam is string;
      }
    }
    
    // Collection: metadata (for connection tracking)
    match /\_metadata/{document=**} {
      allow read, write: if true;
    }
    
    // Deny all other collections
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
